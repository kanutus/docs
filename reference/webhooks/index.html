Webhooks and events

    GET/v1/webhook-endpoints#

    List webhook endpoints

    Permission: webhooks:write

    Response 200

    NameTypeDescription
    data requiredobject[]
    data[].id requiredstring
    data[].url requiredstring
    data[].events requiredstring[]
    data[].status required"active" | "disabled"
    data[].livemode requiredboolean
    data[].created_at requiredstring

    Example

    curl
    curl "https://api.kanutus.com/v1/webhook-endpoints" \
  -H "Authorization: Bearer $KANUTUS_API_KEY"
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/webhook-endpoints", {
  method: "GET",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
  },
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

r = requests.request(
    "GET",
    "https://api.kanutus.com/v1/webhook-endpoints",
    headers={"Authorization": f"Bearer {os.environ['KANUTUS_API_KEY']}"},
)
print(r.status_code, r.json())

    POST/v1/webhook-endpoints#

    Create a webhook endpoint (the signing secret is returned once)

    Permission: webhooks:write

    Request body

    NameTypeDescription
    url requiredstringlength 0–2000
    events required"room.started" | "room.ended" | "room.participant_joined" | "room.participant_left" | "session.started" | "session.ended" | "call.ringing" | "call.answered" | "call.voicemail" | "call.completed" | "call.failed" | "call.blocked" | "call.ended" | "meeting.created" | "meeting.updated" | "meeting.canceled" | "booking.created" | "booking.rescheduled" | "booking.canceled" | "transcript.ready" | "summary.ready" | "voice.consent_completed" | "voice.ready" | "voice.failed" | "usage.threshold" | "credential.expiring" | "academy.completed" | "academy.certified"[]
    descriptionstringlength 0–200

    Response 201

    NameTypeDescription
    id requiredstring
    url requiredstring
    events requiredstring[]
    status required"active" | "disabled"
    livemode requiredboolean
    created_at requiredstring
    secret requiredstringwhsec_… shown only now

    Example

    curl
    curl -X POST "https://api.kanutus.com/v1/webhook-endpoints" \
  -H "Authorization: Bearer $KANUTUS_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"url":"https://example.com","events":["room.started"],"description":"string"}'
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/webhook-endpoints", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({"url":"https://example.com","events":["room.started"],"description":"string"}),
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

r = requests.request(
    "POST",
    "https://api.kanutus.com/v1/webhook-endpoints",
    headers={"Authorization": f"Bearer {os.environ['KANUTUS_API_KEY']}"},
    json={"url":"https://example.com","events":["room.started"],"description":"string"},
)
print(r.status_code, r.json())
