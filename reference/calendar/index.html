    GET/v1/meetings#

    List scheduled meetings

    Permission: calendar:read

    Parameters

    NameTypeDescription
    fromquery · string
    toquery · string
    limitquery · integer

    Response 200

    NameTypeDescription
    data requiredobject[]
    data[].id requiredstring
    data[].url requiredstring
    data[].title requiredstring
    data[].visibility required"private" | "public"
    data[].status required"scheduled" | "live" | "ended"
    data[].created_at requiredstring
    data[].expires_at requiredstring
    data[].starts_at requiredstring | null
    data[].duration_min requiredinteger | null
    data[].invitees requiredstring[]
    data[].langs requiredstring[]
    data[].livemode requiredbooleanfalse for objects created with a kt_test_ key (simulator).

    Example

    curl
    curl "https://api.kanutus.com/v1/meetings" \
  -H "Authorization: Bearer $KANUTUS_API_KEY"
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/meetings", {
  method: "GET",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
  },
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

print(r.status_code, r.json())

    POST/v1/meetings#

    Schedule a meeting (room with date, invitees and languages)

    Permission: calendar:write

    Parameters

    NameTypeDescription
    idempotency-key requiredheader · stringUnique key per create request; retries with the same key return the same result.

    Request body

    NameTypeDescription
    title requiredstringlength 1–120
    starts_at requiredstring
    duration_mininteger5…720 · default 30
    inviteesstring[]
    langsstring[]Languages of the participants, e.g. ["pt", "ja"].
    visibility"private" | "public"default "private"

    Response 201

    NameTypeDescription
    id requiredstring
    url requiredstring
    title requiredstring
    visibility required"private" | "public"
    status required"scheduled" | "live" | "ended"
    created_at requiredstring
    expires_at requiredstring
    starts_at requiredstring | null
    duration_min requiredinteger | null
    invitees requiredstring[]
    langs requiredstring[]
    livemode requiredbooleanfalse for objects created with a kt_test_ key (simulator).

    Example

    curl
    curl -X POST "https://api.kanutus.com/v1/meetings" \
  -H "Authorization: Bearer $KANUTUS_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d '{"title":"Revisão com Kenji","starts_at":"2026-10-12T13:00:00-03:00","duration_min":30,"invitees":[],"langs":[],"visibility":"private"}'
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/meetings", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
    "Idempotency-Key": crypto.randomUUID(),
    "Content-Type": "application/json",
  },
  body: JSON.stringify({"title":"Revisão com Kenji","starts_at":"2026-10-12T13:00:00-03:00","duration_min":30,"invitees":[],"langs":[],"visibility":"private"}),
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

r = requests.request(
    "POST",
    "https://api.kanutus.com/v1/meetings",
    headers={"Authorization": f"Bearer {os.environ['KANUTUS_API_KEY']}", "Idempotency-Key": str(uuid.uuid4())},
    json={"title":"Revisão com Kenji","starts_at":"2026-10-12T13:00:00-03:00","duration_min":30,"invitees":[],"langs":[],"visibility":"private"},
)
print(r.status_code, r.json())

    Request body

    NameTypeDescription
    event_type_idstring
    pagestringBooking page slug, when booking someone else's link
    eventstringEvent slug on that page
    start requiredstring
    guest requiredobject
    guest.name requiredstringlength 1–80
    guest.email requiredstringlength 0–254
    guest.speak requiredstring
    guest.hearstring
    guest.tzstringlength 0–64 · default "UTC"
    guest.localestringlength 0–8 · default "en"
    answersobject

    Response 201

    NameTypeDescription
    id requiredstring
    room_url requiredstring | null
    manage_url requiredstring | null
    start requiredstring
    livemode requiredboolean
