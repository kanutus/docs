Reuniões e agendamento

    GET/v1/meetings#

    Listar reuniões agendadas

    Permissão: calendar:read

    Parâmetros

    NomeTipoDescrição
    fromquery · string
    toquery · string
    limitquery · integer

    Resposta 200

    NomeTipoDescrição
    data obrigatórioobject[]
    data[].id obrigatóriostring
    data[].url obrigatóriostring
    data[].title obrigatóriostring
    data[].visibility obrigatório"private" | "public"
    data[].status obrigatório"scheduled" | "live" | "ended"
    data[].created_at obrigatóriostring
    data[].expires_at obrigatóriostring
    data[].starts_at obrigatóriostring | null
    data[].duration_min obrigatóriointeger | null
    data[].invitees obrigatóriostring[]
    data[].langs obrigatóriostring[]
    data[].livemode obrigatóriobooleanfalse for objects created with a kt_test_ key (simulator).

    Exemplo

    curl
    curl "https://api.kanutus.com/v1/meetings" \
  -H "Authorization: Bearer $KANUTUS_API_KEY"
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/meetings", {
  method: "GET",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
  },
});
console.log(res.status, await res.json());

    POST/v1/meetings#

    Agendar uma reunião (sala com data, convidados e línguas)

    Permissão: calendar:write

    Parâmetros

    NomeTipoDescrição
    idempotency-key obrigatórioheader · stringUnique key per create request; retries with the same key return the same result.

    Corpo da requisição

    NomeTipoDescrição
    title obrigatóriostringlength 1–120
    starts_at obrigatóriostring
    duration_mininteger5…720 · default 30
    inviteesstring[]
    langsstring[]Languages of the participants, e.g. ["pt", "ja"].
    visibility"private" | "public"default "private"

    Resposta 201

    NomeTipoDescrição
    id obrigatóriostring
    url obrigatóriostring
    title obrigatóriostring
    visibility obrigatório"private" | "public"
    status obrigatório"scheduled" | "live" | "ended"
    created_at obrigatóriostring
    expires_at obrigatóriostring
    starts_at obrigatóriostring | null
    duration_min obrigatóriointeger | null
    invitees obrigatóriostring[]
    langs obrigatóriostring[]
    livemode obrigatóriobooleanfalse for objects created with a kt_test_ key (simulator).

    Exemplo

    curl
    curl -X POST "https://api.kanutus.com/v1/meetings" \
  -H "Authorization: Bearer $KANUTUS_API_KEY" \
  -H "Idempotency-Key: $(uuidgen)" \
  -H "Content-Type: application/json" \
  -d '{"title":"Revisão com Kenji","starts_at":"2026-10-12T13:00:00-03:00","duration_min":30,"invitees":[],"langs":[],"visibility":"private"}'
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/meetings", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
    "Idempotency-Key": crypto.randomUUID(),
    "Content-Type": "application/json",
  },
  body: JSON.stringify({"title":"Revisão com Kenji","starts_at":"2026-10-12T13:00:00-03:00","duration_min":30,"invitees":[],"langs":[],"visibility":"private"}),
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

r = requests.request(
    "POST",
    "https://api.kanutus.com/v1/meetings",
    headers={"Authorization": f"Bearer {os.environ['KANUTUS_API_KEY']}", "Idempotency-Key": str(uuid.uuid4())},
    json={"title":"Revisão com Kenji","starts_at":"2026-10-12T13:00:00-03:00","duration_min":30,"invitees":[],"langs":[],"visibility":"private"},
)
print(r.status_code, r.json())
