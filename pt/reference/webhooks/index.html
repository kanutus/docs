Webhooks e eventos

    GET/v1/webhook-endpoints#

    Listar endpoints de webhook

    Permissão: webhooks:write

    Resposta 200

    NomeTipoDescrição
    data obrigatórioobject[]
    data[].id obrigatóriostring
    data[].url obrigatóriostring
    data[].events obrigatóriostring[]
    data[].status obrigatório"active" | "disabled"
    data[].livemode obrigatórioboolean
    data[].created_at obrigatóriostring

    Exemplo

    curl
    curl "https://api.kanutus.com/v1/webhook-endpoints" \
  -H "Authorization: Bearer $KANUTUS_API_KEY"
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/webhook-endpoints", {
  method: "GET",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
  },
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

r = requests.request(
    "GET",
    "https://api.kanutus.com/v1/webhook-endpoints",
    headers={"Authorization": f"Bearer {os.environ['KANUTUS_API_KEY']}"},
)
print(r.status_code, r.json())

    POST/v1/webhook-endpoints#

    Criar endpoint de webhook (o segredo de assinatura aparece uma vez)

    Permissão: webhooks:write

    Corpo da requisição

    NomeTipoDescrição
    url obrigatóriostringlength 0–2000
    events obrigatório"room.started" | "room.ended" | "room.participant_joined" | "room.participant_left" | "session.started" | "session.ended" | "call.ringing" | "call.answered" | "call.voicemail" | "call.completed" | "call.failed" | "call.blocked" | "call.ended" | "meeting.created" | "meeting.updated" | "meeting.canceled" | "booking.created" | "booking.rescheduled" | "booking.canceled" | "transcript.ready" | "summary.ready" | "voice.consent_completed" | "voice.ready" | "voice.failed" | "usage.threshold" | "credential.expiring" | "academy.completed" | "academy.certified"[]
    descriptionstringlength 0–200

    Resposta 201

    NomeTipoDescrição
    id obrigatóriostring
    url obrigatóriostring
    events obrigatóriostring[]
    status obrigatório"active" | "disabled"
    livemode obrigatórioboolean
    created_at obrigatóriostring
    secret obrigatóriostringwhsec_… shown only now

    Exemplo

    curl
    curl -X POST "https://api.kanutus.com/v1/webhook-endpoints" \
  -H "Authorization: Bearer $KANUTUS_API_KEY" \
  -H "Content-Type: application/json" \
  -d '{"url":"https://example.com","events":["room.started"],"description":"string"}'
    Node.js
    const res = await fetch("https://api.kanutus.com/v1/webhook-endpoints", {
  method: "POST",
  headers: {
    Authorization: `Bearer ${process.env.KANUTUS_API_KEY}`,
    "Content-Type": "application/json",
  },
  body: JSON.stringify({"url":"https://example.com","events":["room.started"],"description":"string"}),
});
console.log(res.status, await res.json());
    Python
    import os, uuid, requests

r = requests.request(
    "POST",
    "https://api.kanutus.com/v1/webhook-endpoints",
    headers={"Authorization": f"Bearer {os.environ['KANUTUS_API_KEY']}"},
    json={"url":"https://example.com","events":["room.started"],"description":"string"},
)
print(r.status_code, r.json())
